2-Minute Hardening & Cleanup Playbook

Combine environment flags, OS hosts blocking, offline SQLite compaction, and token revocation.

    ~/.zshrc + /etc/hosts + SQLite VACUUM 
    # 1. Environment opt-out (DO_NOT_TRACK=1 honored from PR #42629 + legacy kill switches)
export DO_NOT_TRACK=1
export T3CODE_TELEMETRY_ENABLED=false
export T3CODE_OTEL_SDK_DISABLED=true
export OTEL_SDK_DISABLED=true

# 2. Block closed-source ingestion gateway in /etc/hosts
sudo sh -c 'printf "\n127.0.0.1 ingest.t3.codes\n::1 ingest.t3.codes\n" >> /etc/hosts'

# 3. Quit T3 Code completely, inspect & compact statev2.sqlite, and remove Antigravity state
ls -lh ~/.t3/userdata/statev2.sqlite*
sqlite3 ~/.t3/userdata/statev2.sqlite "VACUUM;"
rm -rf ~/.t3/userdata/providers/antigravity/ ~/.t3/tools/antigravity-acp/

# 4. Full uninstall cleanup (back up custom files first)
# rm -rf ~/.t3
    PowerShell (Run as Administrator) 
    # 1. Set persistent User & Machine opt-out variables
@("User", "Machine") | ForEach-Object {
  [Environment]::SetEnvironmentVariable("DO_NOT_TRACK", "1", $_)
  [Environment]::SetEnvironmentVariable("T3CODE_TELEMETRY_ENABLED", "false", $_)
  [Environment]::SetEnvironmentVariable("T3CODE_OTEL_SDK_DISABLED", "true", $_)
  [Environment]::SetEnvironmentVariable("OTEL_SDK_DISABLED", "true", $_)
}

# 2. Sinkhole ingest.t3.codes in Windows hosts
$hosts = "$env:SystemRoot\System32\drivers\etc\hosts"
Add-Content -Path $hosts -Value "`r`n127.0.0.1 ingest.t3.codes`r`n::1 ingest.t3.codes"

# 3. Quit T3 Code and clean Antigravity state
Remove-Item -Recurse -Force "$env:USERPROFILE\.t3\userdata\providers\antigravity" -ErrorAction SilentlyContinue
Remove-Item -Recurse -Force "$env:USERPROFILE\.t3\tools\antigravity-acp" -ErrorAction SilentlyContinue
